cmake_minimum_required(VERSION 3.15)
project(DNF90Bootstrap LANGUAGES CXX RC)
set(LAUNCHER_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../Flutter/windows/runner/resources/app_icon.ico")
configure_file(launcher.rc.in "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" @ONLY)
add_executable(DNF90Launcher WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
target_compile_features(DNF90Launcher PRIVATE cxx_std_17)
target_compile_options(DNF90Launcher PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
set_property(TARGET DNF90Launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(DNF90Launcher PRIVATE UNICODE _UNICODE NOMINMAX)
